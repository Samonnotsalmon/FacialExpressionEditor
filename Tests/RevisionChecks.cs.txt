// Run in the open Editor: isuzu-unity-cli call execute_code --file <this file>
// Creates only a uniquely named temporary asset folder, removed in finally.
var checks = new System.Collections.Generic.List<string>();
void Check(bool pass, string name) { if (!pass) throw new Exception(name); checks.Add(name); }
var assembly = typeof(Samon.FacialExpressionEditor.Editor.FxImporter).Assembly;
object Call(string type, string method, params object[] args) => assembly.GetType("Samon.FacialExpressionEditor.Editor." + type).GetMethod(method,
    System.Reflection.BindingFlags.Static | System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic).Invoke(null, args);
var folder = "Assets/__FEE_Verification_" + Guid.NewGuid().ToString("N");
AssetDatabase.CreateFolder("Assets", folder.Substring(7));
var root = new GameObject("Verification avatar");
var face = new GameObject("Face"); face.transform.SetParent(root.transform);
var mesh = new Mesh(); mesh.vertices = new[] {Vector3.zero};
mesh.AddBlendShapeFrame("eye_close",100,new[]{Vector3.up},null,null);
mesh.AddBlendShapeFrame("mouth_smile",100,new[]{Vector3.right},null,null);
var renderer = face.AddComponent<SkinnedMeshRenderer>(); renderer.sharedMesh = mesh;
var temporary = new System.Collections.Generic.List<UnityEngine.Object> {mesh};
try {
 var eye = EditorCurveBinding.FloatCurve("Face",typeof(SkinnedMeshRenderer),"blendShape.eye_close");
 var mouth = EditorCurveBinding.FloatCurve("Face",typeof(SkinnedMeshRenderer),"blendShape.mouth_smile");
 var clip = new AnimationClip {name="Original",frameRate=30};
 AnimationUtility.SetEditorCurve(clip,eye,AnimationCurve.Linear(0,0,1,100));
 AnimationUtility.SetEditorCurve(clip,mouth,AnimationCurve.Linear(0,10,1,40));
 var settings=AnimationUtility.GetAnimationClipSettings(clip); settings.loopTime=true; AnimationUtility.SetAnimationClipSettings(clip,settings);
 AssetDatabase.CreateAsset(clip,folder+"/Original.anim");
 var set=ScriptableObject.CreateInstance<Samon.FacialExpressionEditor.ExpressionSet>(); set.independentClips=true;
 AssetDatabase.CreateFolder(folder,"First"); AssetDatabase.CreateAsset(set,folder+"/First/Face.asset");
 Check(Samon.FacialExpressionEditor.Editor.ExpressionFileImporter.Import(set,new UnityEngine.Object[]{clip,clip})==1,"File import deduplicates sources");
 var expression=set.expressions.Single();
 Check(expression.clip!=clip && expression.originalClip==clip && expression.clip.isLooping,"Imported clip is independent and preserves looping");
 Check(Samon.FacialExpressionEditor.Editor.ExpressionFileImporter.Import(set,new UnityEngine.Object[]{clip})==0,"Repeated import is idempotent");
 var part=new Samon.FacialExpressionEditor.FacialPart{name=expression.clip.name,clip=expression.clip}; set.parts.Add(part);
 Samon.FacialExpressionEditor.Editor.ExpressionSetUtility.MakePartClipEditable(set,part);
 object Usage()=>Call("ClipUsage","Compute",set);
 bool Used(object usage,AnimationClip c)=>(bool)usage.GetType().GetMethod("IsUsed").Invoke(usage,new object[]{c});
 Check(Used(Usage(),expression.clip) && Used(Usage(),clip),"Part assignment marks library expression and original source as used");
 var remembered=part.originalClip; part.originalClip=null;
 Check(Used(Usage(),expression.clip),"Legacy unchanged part copy is recognized without a source reference");
 part.originalClip=remembered;
 AnimationUtility.SetEditorCurve(part.clip,mouth,AnimationCurve.Constant(0,1,65));
 Check(Used(Usage(),expression.clip),"Editing a part does not lose its source assignment");
 set.parts.Remove(part);
 Check(!Used(Usage(),expression.clip),"Removing the only part assignment restores unassigned status");
 set.exclusiveGroups.Add("目の装飾"); set.previewHeight=.08f; set.previewZoom=1.2f;
 var faceHash=(string)Call("ThumbnailKeys","FaceHash",root,set); set.previewHeight=.1f;
 Check(faceHash!=(string)Call("ThumbnailKeys","FaceHash",root,set),"Camera framing changes thumbnail cache identity");
 AssetDatabase.CreateFolder(folder,"Second");
 var copy=Samon.FacialExpressionEditor.Editor.ExpressionSetUtility.Duplicate(set,folder+"/Second/Face.asset");
 Check(copy.exclusiveGroups.Contains("目の装飾") && copy.previewHeight==.1f && copy.previewZoom==1.2f,"Duplicated setup preserves registered groups and camera framing");
 AnimationUtility.SetEditorCurve(copy.expressions[0].clip,mouth,AnimationCurve.Constant(0,1,99));
 Check(AnimationUtility.GetEditorCurve(expression.clip,mouth).Evaluate(1)==40 && AnimationUtility.GetEditorCurve(clip,mouth).Evaluate(1)==40,"Editing duplicated base-face set leaves both originals intact");
 var baseline=ScriptableObject.CreateInstance<Samon.FacialExpressionEditor.FaceVariant>(); temporary.Add(baseline);
 baseline.useCapturedValues=true;
 baseline.baseFace.Add(new Samon.FacialExpressionEditor.BaseFaceKey{path="Face",blendShape="mouth_smile",referenceValue=0,variantValue=20});
 renderer.SetBlendShapeWeight(1,80);
 var preview=(AnimationClip)Call("PreviewClips","ForExpression",expression,baseline,root); temporary.Add(preview);
 Check(Mathf.Approximately(AnimationUtility.GetEditorCurve(preview,mouth).Evaluate(.5f),45),"Captured smile offsets animation without following unrelated scene changes");
 baseline.baseFaceExclusions.Add(new Samon.FacialExpressionEditor.BaseFaceExclusion{expressionId=expression.id,keys=new System.Collections.Generic.List<string>{"Face|mouth_smile"}});
 var excluded=(AnimationClip)Call("PreviewClips","ForExpression",expression,baseline,root); temporary.Add(excluded);
 Check(Mathf.Approximately(AnimationUtility.GetEditorCurve(excluded,mouth).Evaluate(.5f),25),"Excluded smile preserves original motion");
 expression.freezeAnimation=true; expression.freezePosition=.25f;
 var frozen=(AnimationClip)Call("PreviewClips","ForExpression",expression,baseline,root); temporary.Add(frozen);
 Check(Mathf.Approximately(AnimationUtility.GetEditorCurve(frozen,mouth).Evaluate(0),17.5f) && !(bool)Call("PreviewClips","IsTimeVarying",frozen) && !frozen.isLooping,"Freeze samples requested time and disables playback");
 expression.freezeAnimation=false;
 var fist=(AnimationClip)Call("PreviewClips","ForFist",expression,baseline,root,set); temporary.Add(fist);
 Check(AnimationUtility.GetEditorCurve(fist,mouth).Evaluate(0)==40 && AnimationUtility.GetEditorCurve(fist,mouth).Evaluate(1)==40 && AnimationUtility.GetEditorCurve(fist,eye).Evaluate(0)==0 && AnimationUtility.GetEditorCurve(fist,eye).Evaluate(1)==100,"Fist moves eyes while preserving assigned mouth at all squeeze levels");
 var mouthClip=new AnimationClip(); temporary.Add(mouthClip); AnimationUtility.SetEditorCurve(mouthClip,mouth,AnimationCurve.Constant(0,0,70));
 var composed=(AnimationClip)Call("ExpressionComposerWindow","Compose",new AnimationClip[]{clip,mouthClip},new System.Collections.Generic.HashSet<string>[]{new System.Collections.Generic.HashSet<string>(),new System.Collections.Generic.HashSet<string>()}); temporary.Add(composed);
 Check(composed.isLooping && Mathf.Approximately(AnimationUtility.GetEditorCurve(composed,eye).Evaluate(.5f),50) && AnimationUtility.GetEditorCurve(composed,mouth).Evaluate(.5f)==70,"Composition preserves animated eyes and prioritizes selected mouth source");
 var fresh=ScriptableObject.CreateInstance<Samon.FacialExpressionEditor.ExpressionSet>(); fresh.faceMeshPaths.Add("Face");
 AssetDatabase.CreateFolder(folder,"Fresh"); AssetDatabase.CreateAsset(fresh,folder+"/Fresh/Face.asset");
 renderer.SetBlendShapeWeight(0,13);
 var created=Samon.FacialExpressionEditor.Editor.ExpressionSetUtility.CreateExpression(fresh,"Base face",Samon.FacialExpressionEditor.Editor.ExpressionSetUtility.NewExpressionSource.CurrentFace,null,root,baseline);
 var createdPreview=(AnimationClip)Call("PreviewClips","ForExpression",created,baseline,root); temporary.Add(createdPreview);
 Check(AnimationUtility.GetCurveBindings(created.clip).Length==2 && AnimationUtility.GetEditorCurve(createdPreview,eye).Evaluate(0)==13 && AnimationUtility.GetEditorCurve(createdPreview,mouth).Evaluate(0)==20,"New expression captures selected base mesh and applies saved base correction only once");
 var rename=Samon.FacialExpressionEditor.Editor.ExpressionSetUtility.RenameExpression(set,expression,"Renamed");
 Check(string.IsNullOrEmpty(rename) && expression.clip.name=="Renamed" && AssetDatabase.GetAssetPath(expression.clip).EndsWith("/Renamed.anim") && clip.name=="Original","Renaming changes owned asset and preserves source name");
 var duplicateName=Samon.FacialExpressionEditor.Editor.ExpressionSetUtility.CreateExpression(fresh,"Base face",Samon.FacialExpressionEditor.Editor.ExpressionSetUtility.NewExpressionSource.CurrentFace,null,root,baseline);
 Check(duplicateName.name==duplicateName.clip.name && duplicateName.name==System.IO.Path.GetFileNameWithoutExtension(AssetDatabase.GetAssetPath(duplicateName.clip)) && duplicateName.name!=created.name,"Duplicate names use identical unique display and file names");
 created.name="Custom display";
 fresh.independentClips=true;
 Samon.FacialExpressionEditor.Editor.ExpressionSetUtility.SynchronizeExpressionNames(fresh);
 Check(created.name=="Custom display" && created.clip.name==created.name && AssetDatabase.GetAssetPath(created.clip).EndsWith("/Custom display.anim"),"Existing display name is retained and synchronized to owned file");
 var unused=new Samon.FacialExpressionEditor.Expression{name="Unused"}; fresh.expressions.Add(unused);
 var gestureSet=new Samon.FacialExpressionEditor.GestureSet{name="Core"}; gestureSet.mapping.left[1]=created.id; gestureSet.mapping.right[2]=created.id; fresh.gestureSets.Add(gestureSet);
 fresh.menu.Add(new Samon.FacialExpressionEditor.MenuNode{kind=Samon.FacialExpressionEditor.MenuNodeKind.Expression,expressionId=duplicateName.id});
 var assigned=Samon.FacialExpressionEditor.Editor.ExpressionSetUtility.AssignedExpressions(fresh);
 Check(assigned.Count==2 && assigned.Contains(created) && assigned.Contains(duplicateName) && !assigned.Contains(unused),"Control list includes assigned gesture and fixed expressions once and excludes unused expressions");
 gestureSet.mapping.left[1]=null; gestureSet.mapping.right[2]=null;
 Check(!Samon.FacialExpressionEditor.Editor.ExpressionSetUtility.AssignedExpressions(fresh).Contains(created),"Removing assignments removes expression from control list");
 var newPart=Samon.FacialExpressionEditor.Editor.ExpressionSetUtility.CreatePart(fresh,"新しいパーツ");
 var newPartId=newPart.id; var newPartGuid=AssetDatabase.AssetPathToGUID(AssetDatabase.GetAssetPath(newPart.clip));
 Check(newPart.name==newPart.clip.name && AssetDatabase.GetAssetPath(newPart.clip).EndsWith("/"+newPart.name+".anim"),"New part display, clip and file names match");
 var partRename=Samon.FacialExpressionEditor.Editor.ExpressionSetUtility.RenamePart(fresh,newPart,"頬染め");
 Check(string.IsNullOrEmpty(partRename) && newPart.name=="頬染め" && newPart.clip.name==newPart.name && AssetDatabase.GetAssetPath(newPart.clip).EndsWith("/頬染め.anim") && newPart.id==newPartId && AssetDatabase.AssetPathToGUID(AssetDatabase.GetAssetPath(newPart.clip))==newPartGuid,"Part rename synchronizes names and preserves references");
 var sameNamePart=Samon.FacialExpressionEditor.Editor.ExpressionSetUtility.CreatePart(fresh,"頬染め");
 Check(sameNamePart.name!=newPart.name && sameNamePart.name==sameNamePart.clip.name && AssetDatabase.GetAssetPath(sameNamePart.clip).EndsWith("/"+sameNamePart.name+".anim"),"Duplicate part names stay aligned with unique files");
 Check(!string.IsNullOrEmpty(Samon.FacialExpressionEditor.Editor.ExpressionSetUtility.RenamePart(fresh,newPart,"bad/name")) && newPart.name=="頬染め" && newPart.clip.name==newPart.name,"Invalid part name leaves names unchanged");
 var importedPart=new Samon.FacialExpressionEditor.FacialPart{name="Imported",clip=clip}; fresh.parts.Add(importedPart);
 var originalPath=AssetDatabase.GetAssetPath(clip); var originalName=clip.name;
 var importedRename=Samon.FacialExpressionEditor.Editor.ExpressionSetUtility.RenamePart(fresh,importedPart,"汗");
 Check(string.IsNullOrEmpty(importedRename) && importedPart.clip!=clip && importedPart.originalClip==clip && importedPart.name=="汗" && importedPart.clip.name=="汗" && AssetDatabase.GetAssetPath(importedPart.clip).EndsWith("/汗.anim") && AssetDatabase.GetAssetPath(clip)==originalPath && clip.name==originalName,"Imported part renames an editable copy and preserves source");
 return new {passed=checks.Count,checks};
} finally {
 UnityEngine.Object.DestroyImmediate(root);
 foreach(var obj in temporary) if(obj!=null) UnityEngine.Object.DestroyImmediate(obj);
 AssetDatabase.DeleteAsset(folder);
}
