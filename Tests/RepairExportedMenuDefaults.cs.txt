int Fix(Samon.FacialExpressionEditor.FacialExpressionAvatar owner,bool undo){int changed=0;if(!owner.editableMenu)return 0;foreach(var marker in owner.editableMenu.GetComponentsInChildren<Samon.FacialExpressionEditor.EditableMenuItem>(true)){var item=marker.GetComponent<nadena.dev.modular_avatar.core.ModularAvatarMenuItem>();if(!item||item.Control==null)continue;var initial=item.Control.parameter?.name=="FEE/Mode"&&item.Control.value==0;if(item.isDefault==initial&&!item.isSaved&&!item.automaticValue)continue;if(undo)Undo.RecordObject(item,"FEE初期値を修正");item.isDefault=initial;item.isSaved=false;item.automaticValue=false;EditorUtility.SetDirty(item);if(undo)PrefabUtility.RecordPrefabInstancePropertyModifications(item);changed++;}return changed;}
