// Run with isuzu-unity-cli execute_code in EditorTest, which has these avatar packages.
var paths = new[]{
 "Assets/Shinano/Prefab/Shinano.prefab", "Assets/MANUKA/Prefab/MANUKA_lilToon.prefab",
 "Assets/onair/Alue/Prefab/Alue.prefab", "Assets/Kogomi/Kogomi.prefab",
 "Assets/STUDIO_MOCA/NagiyaRuri/Prefab/Avatar_Simple/NagiyaRuri_Simple.prefab",
 "Assets/STUDIO_MOCA/NagiyaRuri/_ModuleAssets/Prefab_MA_Avatar_Simple/NagiyaRuri_Modular_Simple.prefab"};
var results = new System.Collections.Generic.List<object>();
var folder="Assets/__FEE_BuildCheck_"+Guid.NewGuid().ToString("N");
AssetDatabase.CreateFolder("Assets",folder.Substring(7));
var previousFilter=Debug.unityLogger.filterLogType;
try {
 Debug.unityLogger.filterLogType=LogType.Warning;
 for(var i=0;i<paths.Length;i++) {
  var prefab=AssetDatabase.LoadAssetAtPath<GameObject>(paths[i]); if(prefab==null) throw new Exception("Missing fixture: "+paths[i]);
  var avatar=UnityEngine.Object.Instantiate(prefab); avatar.name="FEE build verification";
  UnityEngine.Object temporaryMenu=null;
  try {
   var descriptor=avatar.GetComponent<VRC.SDK3.Avatars.Components.VRCAvatarDescriptor>();
   var fxSource=Samon.FacialExpressionEditor.Editor.FxImporter.GetFx(descriptor);
   var source=fxSource.animationClips.First(c=>AnimationUtility.GetCurveBindings(c).Any(b=>b.propertyName.StartsWith("blendShape.")));
   AssetDatabase.CreateFolder(folder,i.ToString());
   var set=ScriptableObject.CreateInstance<Samon.FacialExpressionEditor.ExpressionSet>();
   set.menuIcons=false; set.independentClips=true; set.faceControlImported=true; set.explicitMenuSelection=true;
   AssetDatabase.CreateAsset(set,folder+"/"+i+"/Face.asset");
   Samon.FacialExpressionEditor.Editor.ExpressionFileImporter.Import(set,new UnityEngine.Object[]{source});
   set.originalGestureLayers=Samon.FacialExpressionEditor.Editor.FxImporter.PeekGestures(descriptor).layers;
   var gesture=new Samon.FacialExpressionEditor.GestureSet{name="Core"}; gesture.mapping.right[2]=set.expressions[0].id;
   set.gestureSets.Add(gesture);
   set.gestureSets.Add(new Samon.FacialExpressionEditor.GestureSet{name="Core 2"});
   Samon.FacialExpressionEditor.Editor.ExpressionSetUtility.AddMenuFromSets(set);
   if(descriptor.expressionsMenu!=null) {
    var menu=UnityEngine.Object.Instantiate(descriptor.expressionsMenu); temporaryMenu=menu;
    menu.controls.Add(new VRC.SDK3.Avatars.ScriptableObjects.VRCExpressionsMenu.Control{name="FEE selected menu removal",type=VRC.SDK3.Avatars.ScriptableObjects.VRCExpressionsMenu.Control.ControlType.Button});
    set.replacedMenuItems.Add(new Samon.FacialExpressionEditor.OriginalMenuSelection{menu=menu,index=menu.controls.Count-1,controlName="FEE selected menu removal"});
    descriptor.expressionsMenu=menu;
   }
   avatar.AddComponent<Samon.FacialExpressionEditor.FacialExpressionAvatar>().expressionSet=set;
   nadena.dev.ndmf.AvatarProcessor.ProcessAvatar(avatar);
   var fx=(UnityEditor.Animations.AnimatorController)descriptor.baseAnimationLayers.First(l=>l.type==VRC.SDK3.Avatars.Components.VRCAvatarDescriptor.AnimLayerType.FX).animatorController;
   if(!fx.layers.Any(l=>l.name=="[FacialExpressionEditor] Expression")) throw new Exception("Missing generated expression layer: " + paths[i]);
   if(!fx.layers.Any(l=>l.name=="[FacialExpressionEditor] Blink")) throw new Exception("Missing mandatory blink replacement: " + paths[i]);
   if(fx.layers.Any(l=>set.originalGestureLayers.Contains(l.name))) throw new Exception("Original gesture layers were not replaced");
   var visited=new System.Collections.Generic.HashSet<VRC.SDK3.Avatars.ScriptableObjects.VRCExpressionsMenu>();
   var names=new System.Collections.Generic.List<string>();
   void ReadMenu(VRC.SDK3.Avatars.ScriptableObjects.VRCExpressionsMenu menu) {if(menu==null || !visited.Add(menu))return;foreach(var c in menu.controls){names.Add(c.name);ReadMenu(c.subMenu);}}
   ReadMenu(descriptor.expressionsMenu);
   if(names.Contains("FEE selected menu removal") || !names.Contains("固定解除")) throw new Exception("Menu removal/reset check failed");
   if(!names.Contains("まばたきOFF") || !names.Contains("ダンスギミック有効（MMD対応）")) throw new Exception("Missing settings menu toggles");
   foreach(var key in new[]{"FEE/BlinkOff","FEE/DanceEnabled"}) {
    var param=descriptor.expressionParameters.parameters.Single(p=>p.name==key);
    if(param.valueType!=VRC.SDK3.Avatars.ScriptableObjects.VRCExpressionParameters.ValueType.Bool || param.defaultValue!=0 || param.saved || !param.networkSynced) throw new Exception("Incorrect toggle parameter: "+key);
   }
   var danceMenu=fx.layers.Single(l=>l.name=="[FacialExpressionEditor] Dance menu control");
   var normal=danceMenu.stateMachine.states.Single(s=>s.state.name=="Normal").state;
   var active=danceMenu.stateMachine.states.Single(s=>s.state.name=="MMD station").state;
   if(normal.transitions.Single().conditions.Length!=2 || !normal.transitions.Single().conditions.Any(c=>c.parameter=="InStation" && c.mode==UnityEditor.Animations.AnimatorConditionMode.If)) throw new Exception("Dance entry must require toggle AND station");
   if(active.transitions.Length!=2 || active.transitions.Any(t=>t.conditions.Length!=1 || t.conditions[0].mode!=UnityEditor.Animations.AnimatorConditionMode.IfNot)) throw new Exception("Dance must exit on toggle off OR station exit");
   foreach(var item in new[]{(state:normal,weight:1f),(state:active,weight:0f)}) {
    var control=item.state.behaviours.OfType<VRC.SDK3.Avatars.Components.VRCPlayableLayerControl>().Single();
    if(control.layer!=VRC.SDKBase.VRC_PlayableLayerControl.BlendableLayer.FX || control.goalWeight!=item.weight) throw new Exception("Wrong FX playable weight");
    if(item.state.behaviours.OfType<VRC.SDK3.Avatars.Components.VRCAvatarParameterDriver>().SelectMany(d=>d.parameters).Any(p=>p.name!="FEE/DanceActive")) throw new Exception("Dance must preserve user selections");
   }
   var blinkGate=fx.layers.Single(l=>l.name=="[FacialExpressionEditor] Motion gate [FacialExpressionEditor] Blink");
   var resume=blinkGate.stateMachine.states.Single(s=>s.state.name=="Yield").state.transitions.Single();
   if(!resume.conditions.Any(c=>c.parameter=="FEE/BlinkOff" && c.mode==UnityEditor.Animations.AnimatorConditionMode.IfNot) || !resume.conditions.Any(c=>c.parameter=="FEE/DanceActive" && c.mode==UnityEditor.Animations.AnimatorConditionMode.IfNot)) throw new Exception("Blink must wait for both menu overrides to clear");
   var dance=fx.layers.FirstOrDefault(l=>l.name.StartsWith("[FacialExpressionEditor] Motion rule"));
   if(paths[i].Contains("NagiyaRuri")) {
    if(dance==null || !fx.animationClips.Any(c=>c.name=="NGR_Holiday_FX")) throw new Exception("Dance face was lost");
    var release=fx.layers.First(l=>l.name=="[FacialExpressionEditor] Emote Release");
    if(release.stateMachine.states.Any(s=>s.state.name.Contains("suspend"))) throw new Exception("Dance would clear fixed expression on resume");
    if(!fx.layers.First(l=>l.name=="[FacialExpressionEditor] Expression").stateMachine.states.Any(s=>s.state.name.Contains("suspend"))) throw new Exception("Missing dance suspension");
   }
   results.Add(new{avatar=paths[i],passed=true,fxLayers=fx.layers.Length,danceProtected=dance!=null});
  } finally {UnityEngine.Object.DestroyImmediate(avatar); if(temporaryMenu!=null && !AssetDatabase.Contains(temporaryMenu))UnityEngine.Object.DestroyImmediate(temporaryMenu);}
 }
 return results;
} finally {Debug.unityLogger.filterLogType=previousFilter; AssetDatabase.DeleteAsset(folder);}
