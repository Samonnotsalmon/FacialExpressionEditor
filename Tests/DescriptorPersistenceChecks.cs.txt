// Reproduce direct SDK inspector assignment on a prefab instance, then save/reload a temporary scene.
var folder="Assets/__FEE_DescriptorCheck_"+Guid.NewGuid().ToString("N");
AssetDatabase.CreateFolder("Assets",folder.Substring(7));
var activeScene=SceneManager.GetActiveScene();
var scene=EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Additive);
var type=typeof(Samon.FacialExpressionEditor.Editor.FxImporter).Assembly.GetType("Samon.FacialExpressionEditor.Editor.AvatarDescriptorPersistence");
var checks=new System.Collections.Generic.List<string>();
void Check(bool condition,string name){if(!condition)throw new Exception(name);checks.Add(name);}
try {
 var source=new GameObject("Avatar");
 var descriptor=source.AddComponent<VRC.SDK3.Avatars.Components.VRCAvatarDescriptor>();
 var body=new GameObject("Body");body.transform.SetParent(source.transform,false);
 var owl=new GameObject("Owl");owl.transform.SetParent(source.transform,false);
 descriptor.VisemeSkinnedMesh=body.AddComponent<SkinnedMeshRenderer>();
 owl.AddComponent<SkinnedMeshRenderer>();
 descriptor.lipSync=VRC.SDKBase.VRC_AvatarDescriptor.LipSyncStyle.VisemeBlendShape;
 descriptor.VisemeBlendShapes=new[]{"BodyAA"};
 var prefab=PrefabUtility.SaveAsPrefabAsset(source,folder+"/Avatar.prefab");
 UnityEngine.Object.DestroyImmediate(source);
 var instance=(GameObject)PrefabUtility.InstantiatePrefab(prefab,scene);
 descriptor=instance.GetComponent<VRC.SDK3.Avatars.Components.VRCAvatarDescriptor>();
 descriptor.VisemeSkinnedMesh=instance.transform.Find("Owl").GetComponent<SkinnedMeshRenderer>();
 descriptor.VisemeBlendShapes=new[]{"OwlAA","OwlEE"};
 descriptor.ViewPosition=new Vector3(0,1.7f,.1f);
 Check(!new SerializedObject(descriptor).FindProperty("VisemeSkinnedMesh").prefabOverride,"Reproduces unrecorded direct SDK assignment");
 type.GetMethod("Observe").Invoke(null,new object[]{instance});
 Check(new SerializedObject(descriptor).FindProperty("VisemeSkinnedMesh").prefabOverride,"Registers current accessory mesh as a prefab override");
 descriptor.VisemeBlendShapes=new[]{"OwlAA","OwlEE","OwlOH"};
 var path=folder+"/Fixture.unity";
 EditorSceneManager.SaveScene(scene,path);
 EditorSceneManager.CloseScene(scene,true);
 scene=EditorSceneManager.OpenScene(path,OpenSceneMode.Additive);
 descriptor=scene.GetRootGameObjects().Single().GetComponent<VRC.SDK3.Avatars.Components.VRCAvatarDescriptor>();
 Check(descriptor.VisemeSkinnedMesh.name=="Owl","Mesh survives scene save/reload");
 Check(descriptor.VisemeBlendShapes.SequenceEqual(new[]{"OwlAA","OwlEE","OwlOH"}),"Later inspector edits are flushed before saving");
 Check(descriptor.lipSync==VRC.SDKBase.VRC_AvatarDescriptor.LipSyncStyle.VisemeBlendShape,"Lip-sync mode survives scene save/reload");
 Check(descriptor.ViewPosition==new Vector3(0,1.7f,.1f),"Other user descriptor settings remain intact");
 var original=prefab.GetComponent<VRC.SDK3.Avatars.Components.VRCAvatarDescriptor>();
 Check(original.VisemeSkinnedMesh.name=="Body"&&original.VisemeBlendShapes.SequenceEqual(new[]{"BodyAA"}),"Source prefab is never changed");
 return checks;
}finally{
 SceneManager.SetActiveScene(activeScene);
 if(scene.IsValid())EditorSceneManager.CloseScene(scene,true);
 AssetDatabase.DeleteAsset(folder);
}
